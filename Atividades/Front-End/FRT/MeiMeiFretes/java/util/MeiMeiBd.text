CREATE TABLE usuario (
    id_usuario INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    cpf VARCHAR(11) NOT NULL UNIQUE,
    tipo VARCHAR(20) NOT NULL,

    CHECK (tipo IN ('CLIENTE', 'FORNECEDOR'))
);

CREATE TABLE localizacao (
    id_localizacao INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    cidade VARCHAR(100) NOT NULL,
    estado VARCHAR(100) NOT NULL,
    codigo VARCHAR(20)
);

CREATE TABLE encomenda (
    id_encomenda INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    codigo_rastreamento BIGINT GENERATED ALWAYS AS IDENTITY UNIQUE,
    nome VARCHAR(100) NOT NULL,
    descricao TEXT,
    id_fornecedor INTEGER NOT NULL,
    id_origem INTEGER NOT NULL,
    id_destino INTEGER NOT NULL,
    id_localizacao_atual INTEGER NOT NULL,
    data_criacao VARCHAR(20) NOT NULL,

    FOREIGN KEY (id_fornecedor)
        REFERENCES usuario(id_usuario),

    FOREIGN KEY (id_origem)
        REFERENCES localizacao(id_localizacao),

    FOREIGN KEY (id_destino)
        REFERENCES localizacao(id_localizacao),

    FOREIGN KEY (id_localizacao_atual)
        REFERENCES localizacao(id_localizacao)
);

CREATE TABLE tipo_status (
    id_tipo_status INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nome VARCHAR(50) NOT NULL UNIQUE,
    descricao TEXT
);

INSERT INTO tipo_status (nome, descricao) VALUES
('RECEBIDA', 'Encomenda recebida'),
('EM_TRANSITO', 'Encomenda sendo transportada'),
('EM_DISTRIBUICAO', 'Encomenda em processo de distribuição'),
('ENTREGUE', 'Encomenda entregue ao destinatário');

CREATE TABLE status_encomenda (
    id_status INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_encomenda INTEGER NOT NULL UNIQUE,
    id_tipo_status INTEGER NOT NULL,
    descricao TEXT,
    data VARCHAR(20) NOT NULL,

    FOREIGN KEY (id_encomenda)
        REFERENCES encomenda(id_encomenda),

    FOREIGN KEY (id_tipo_status)
        REFERENCES tipo_status(id_tipo_status)
);

CREATE TABLE historico (
    id_historico INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_encomenda INTEGER NOT NULL,
    id_tipo_status INTEGER NOT NULL,
    id_localizacao INTEGER NOT NULL,
    descricao TEXT,
    data VARCHAR(20) NOT NULL,

    FOREIGN KEY (id_encomenda)
        REFERENCES encomenda(id_encomenda),

    FOREIGN KEY (id_tipo_status)
        REFERENCES tipo_status(id_tipo_status),

    FOREIGN KEY (id_localizacao)
        REFERENCES localizacao(id_localizacao)
);
